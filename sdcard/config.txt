#
# BMMacintosh - bare metal 68k Macintosh emulator for the Raspberry Pi:
# the Mac Plus (every Pi) and the Mac II and Quadra (Pi 4 and 5)
#
# Raspberry Pi firmware settings. Emulator options are in cmdline.txt.
#

# Each Mac is a kernel of its own: macplus*.img the Mac Plus (umac),
# macii*.img the Mac II and Quadra (Basilisk II). BMMacintosh's menu (F12,
# Machine) switches between them by changing the kernel= lines of the
# [pi4] and [pi5] sections below, and restarting.
# They run in 32 bit mode on the Pi Zero, 2 and 3, and in 64 bit mode on
# the Pi 4 and 5.
arm_64bit=0
# [pi0] also matches the Zero 2 W, which [pi02] then sets to the Pi 3's.
[pi0]
kernel=macplus.img
[pi2]
kernel=macplus7.img
[pi3]
kernel=macplus8-32.img
[pi02]
kernel=macplus8-32.img
# The Pi 4 B and Pi 400 (both match [pi4]). As in Circle's boot/config64.txt:
# 64 bit mode, the kernel loaded at 0x80000, and Circle's ARM stub.
[pi4]
arm_64bit=1
kernel_address=0x80000
kernel=macii8-rpi4.img
armstub=armstub8-rpi4.bin
# The Pi 5, Pi 500 and Pi 500+ (all match [pi5]). As in Circle's
# boot/config64.txt; the Pi 5 needs no ARM stub. Its firmware sets the
# framebuffer's depth from here rather than as asked: 32 bit.
[pi5]
arm_64bit=1
kernel_address=0x80000
kernel=macii_2712.img
framebuffer_depth=32
framebuffer_ignore_alpha=1
[all]

# Video: 1280x720 at 60 Hz, the Mac's rate (60.15 Hz) near enough: the
# emulator's frames are locked to the display when its refresh rate is
# that close, which gives smooth scrolling. On the Pi 4, Pi 400 and Pi 5
# BMMacintosh sets the display to the Mac's exact rate itself (see
# docs/PI4.md); hdmi_mode only chooses the resolution on the Pi 4 (the
# Pi 5's firmware uses the display's preferred one; choose it under
# Video, Resolution).
disable_overscan=1
hdmi_group=1
hdmi_mode=4

# The same for the Pi 4's second HDMI port (HDMI1), which the lines above
# don't cover.
[pi4]
hdmi_group:1=1
hdmi_mode:1=4
[all]

# Other useful modes (hdmi_group=1):
#   hdmi_mode=19  720p  50 Hz        hdmi_mode=4   720p  60 Hz
#   hdmi_mode=31  1080p 50 Hz        hdmi_mode=16  1080p 60 Hz
# The CRT shader looks best at 1080p, especially with curvature; sharp
# pixels then have black bars above and below.

# Uncomment if you get no picture
#hdmi_safe=1

# Uncomment to force HDMI (rather than DVI) mode, for sound over HDMI on
# some monitors
#hdmi_drive=2

# Uncomment to boost the HDMI signal if you get interference
#config_hdmi_boost=4

# Composite video (the 3.5mm jack) instead of HDMI: sdtv_mode=0 is NTSC,
# sdtv_mode=2 is PAL. Unplug HDMI to use it.
sdtv_mode=0

# Scaling filter: 8 = sharp pixels, 2 = soft, 6 = softer (default)
#scaling_kernel=8

gpu_mem=64
disable_splash=1
boot_delay=0
